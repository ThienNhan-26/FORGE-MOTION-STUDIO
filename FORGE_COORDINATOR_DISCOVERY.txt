        text = content.strip()
        text = re.sub(r"^```(?:json)?\s*", "", text, flags=re.I)

                raise PlannerError(
                    f"Task thá»© {index} khÃ´ng pháº£i object JSON."

            if not task_id or task_id in used_ids:

            if skill_name not in allowed:

                raise PlannerError(

from __future__ import annotations
import threading
import uuid
from agents.planner import PlannerAgent
from core.scheduler import TaskScheduler
from core.skill_registry import SkillRegistry
from core.task import Task, TaskStatus
from storage.database import init_db, save_mission, save_task, emit_event
from config import FORGE_MAX_ACTIVE_MISSIONS


class Coordinator:
    _mission_semaphore = threading.Semaphore(FORGE_MAX_ACTIVE_MISSIONS)
    def __init__(self, planner: PlannerAgent | None = None):
        init_db()
        SkillRegistry.discover_skills()
        self.planner = planner or PlannerAgent()
        self.scheduler = TaskScheduler()
        self._threads: dict[str, threading.Thread] = {}
        interrupted = 0
        try:
            from storage.database import recover_interrupted_tasks
            interrupted = recover_interrupted_tasks()
        except Exception:
            pass
        if interrupted:
            emit_event(f"ÄÃ¡nh dáº¥u {interrupted} task bá»‹ giÃ¡n Ä‘oáº¡n tá»« phiÃªn trÆ°á»›c; khÃ´ng tá»± cháº¡y láº¡i side-effect.", "WARN")

    def submit_mission(self, user_request: str) -> str:
        request = user_request.strip()
        if not request:
            raise ValueError("Nhiá»‡m vá»¥ khÃ´ng Ä‘Æ°á»£c Ä‘á»ƒ trá»‘ng.")
        mission_id = uuid.uuid4().hex[:10]
        save_mission(mission_id, request, "QUEUED")
        emit_event(f"ÄÃ£ nháº­n mission {mission_id}: {request}", "INFO", mission_id)
        thread = threading.Thread(target=self._run_mission, args=(mission_id, request), daemon=True, name=f"forge-mission-{mission_id}")
        self._threads[mission_id] = thread
        thread.start()
        return mission_id

    def execute_sync(self, user_request: str) -> tuple[str, list[Task], bool]:
        request = user_request.strip()
        mission_id = uuid.uuid4().hex[:10]
        save_mission(mission_id, request, "PLANNING")
        tasks = self._plan_tasks(mission_id, request)
        success = self.scheduler.run_graph(tasks, mission_id)
        status = "COMPLETED" if success else "FAILED"
        save_mission(mission_id, request, status, f"{sum(t.status == TaskStatus.COMPLETED for t in tasks)}/{len(tasks)} task hoÃ n thÃ nh")
        return mission_id, tasks, success

    def _plan_tasks(self, mission_id: str, request: str) -> list[Task]:
        emit_event("Planner Ä‘ang phÃ¢n tÃ­ch yÃªu cáº§u vÃ  xÃ¢y dá»±ng DAG...", "INFO", mission_id)
        task_data = self.planner.create_task_graph(request)
        # Planner IDs are local to a mission (often t1/t2); namespace them globally for SQLite.
        id_map = {str(item["task_id"]): f"{mission_id}_{item['task_id']}" for item in task_data}
        tasks = [Task(
            task_id=id_map[str(item["task_id"])], title=str(item["title"]),
            description=str(item.get("description", "")), task_type=str(item.get("task_type", "GENERAL")),
            skill_name=str(item["skill_name"]),
            dependencies=[id_map[str(dep)] for dep in item.get("dependencies", [])],
            input_data=dict(item.get("input_data", {})), mission_id=mission_id,
        ) for item in task_data]
        for task in tasks:
            save_task(task)
        emit_event(f"Planner Ä‘Ã£ táº¡o {len(tasks)} task: " + ", ".join(t.task_id for t in tasks), "INFO", mission_id)
        return tasks

    def _run_mission(self, mission_id: str, request: str) -> None:
        with self._mission_semaphore:
            save_mission(mission_id, request, "PLANNING")
            try:
                tasks = self._plan_tasks(mission_id, request)
                success = self.scheduler.run_graph(tasks, mission_id)
                status = "COMPLETED" if success else "FAILED"
                summary = f"{sum(t.status == TaskStatus.COMPLETED for t in tasks)}/{len(tasks)} task hoÃ n thÃ nh"
                save_mission(mission_id, request, status, summary)
                emit_event(f"Mission {mission_id} káº¿t thÃºc: {status} ({summary}).", "SUCCESS" if success else "ERROR", mission_id)
            except Exception as exc:
                save_mission(mission_id, request, "FAILED", str(exc))
                emit_event(f"Mission {mission_id} lá»—i: {exc}", "ERROR", mission_id)
            finally:

from __future__ import annotations
import json
import threading
import urllib.error
import urllib.request
from config import OLLAMA_BASE_URL, OLLAMA_MODEL, OLLAMA_TIMEOUT


class OllamaError(RuntimeError):
    pass


class OllamaClient:
    _request_lock = threading.Semaphore(1)  # Avoid competing large model loads on local hardware.
    def __init__(self, model: str | None = None, timeout: int | None = None):
        self.model = model or OLLAMA_MODEL
        self.timeout = timeout or OLLAMA_TIMEOUT
        self.url = f"{OLLAMA_BASE_URL}/api/chat"

    def chat(self, messages: list[dict[str, str]], temperature: float = 0.1, json_mode: bool = False) -> str:
        payload: dict = {
            "model": self.model,
            "messages": messages,
            "stream": False,
            "options": {"temperature": temperature},
        }
        if json_mode:
            payload["format"] = "json"
        request = urllib.request.Request(
            self.url,
            data=json.dumps(payload, ensure_ascii=False).encode("utf-8"),
            headers={"Content-Type": "application/json"},
            method="POST",
        )
        try:
            with self._request_lock:
                with urllib.request.urlopen(request, timeout=self.timeout) as response:
                    data = json.loads(response.read().decode("utf-8"))
        except urllib.error.HTTPError as exc:
            details = exc.read().decode("utf-8", errors="replace")[:1200]
            raise OllamaError(f"Ollama tráº£ HTTP {exc.code} cho model '{self.model}': {details}") from exc
        except urllib.error.URLError as exc:
            reason = getattr(exc, "reason", exc)
            raise OllamaError(
                f"KhÃ´ng káº¿t ná»‘i Ä‘Æ°á»£c Ollama táº¡i {OLLAMA_BASE_URL}. "
                f"HÃ£y má»Ÿ Ollama vÃ  thá»­ láº¡i. Chi tiáº¿t: {reason}"
            ) from exc
        except TimeoutError as exc:
            raise OllamaError(f"Ollama quÃ¡ thá»i gian chá» ({self.timeout}s).") from exc
        except Exception as exc:
            raise OllamaError(f"Pháº£n há»“i Ollama khÃ´ng há»£p lá»‡: {exc}") from exc
        content = data.get("message", {}).get("content")
        if not isinstance(content, str) or not content.strip():
            raise OllamaError("Ollama tráº£ vá» ná»™i dung rá»—ng.")
        return content.strip()

class TaskScheduler:
    def __init__(self, event_sink: Callable | None = None, max_workers: int | None = None):
        self.event_sink = event_sink
        self.max_workers = max_workers or FORGE_MAX_WORKERS

    def _emit(self, message: str, level: str, mission_id: str, task_id: str = "") -> None:
        emit_event(message, level, mission_id, task_id)
        if self.event_sink:
            try:
                self.event_sink(message, level, mission_id, task_id)
            except Exception:
                pass

    def run_graph(self, tasks: list[Task], mission_id: str = "") -> bool:
        by_id = {task.task_id: task for task in tasks}
        if len(by_id) != len(tasks):
            raise ValueError("Task graph cÃ³ task_id trÃ¹ng nhau.")
        for task in tasks:
            task.mission_id = mission_id or task.mission_id
            unknown = set(task.dependencies) - set(by_id)
            if unknown:
                raise ValueError(f"Task {task.task_id} cÃ³ dependency khÃ´ng tá»“n táº¡i: {', '.join(sorted(unknown))}")
            save_task(task)
        pending = set(by_id)
        failed_any = False
        self._emit(f"Scheduler báº¯t Ä‘áº§u: {len(tasks)} task, tá»‘i Ä‘a {self.max_workers} luá»“ng song song.", "INFO", mission_id)

        while pending:
            # Block downstream nodes as soon as one dependency has failed/blocked.
            for task_id in list(pending):
                task = by_id[task_id]
                dependency_states = [by_id[d].status for d in task.dependencies]
                if any(state in {TaskStatus.FAILED, TaskStatus.BLOCKED, TaskStatus.WAITING_APPROVAL} for state in dependency_states):
                    task.status = TaskStatus.BLOCKED
                    task.error = "Má»™t dependency khÃ´ng hoÃ n thÃ nh thÃ nh cÃ´ng."
                    save_task(task)
                    pending.remove(task_id)
                    failed_any = True
                    self._emit(f"Task {task.task_id} bá»‹ BLOCKED vÃ¬ dependency tháº¥t báº¡i.", "WARN", mission_id, task.task_id)

            ready = [by_id[task_id] for task_id in pending
                     if all(by_id[d].status == TaskStatus.COMPLETED for d in by_id[task_id].dependencies)]
            if not ready:
                if pending:
                    for task_id in list(pending):
                        task = by_id[task_id]
                        task.status = TaskStatus.BLOCKED
                        task.error = "Dependency graph bá»‹ chu trÃ¬nh hoáº·c khÃ´ng thá»ƒ giáº£i quyáº¿t."
                        save_task(task)
                        pending.remove(task_id)
                        failed_any = True
                    self._emit("PhÃ¡t hiá»‡n chu trÃ¬nh hoáº·c graph khÃ´ng thá»ƒ tiáº¿n triá»ƒn; cÃ¡c task cÃ²n láº¡i bá»‹ BLOCKED.", "ERROR", mission_id)
                break

            with ThreadPoolExecutor(max_workers=self.max_workers, thread_name_prefix="forge-task") as pool:
                futures = {pool.submit(self._execute_task, task, mission_id): task for task in ready}
                for future in as_completed(futures):
                    task = futures[future]
                    try:
                        success = future.result()
                    except Exception as exc:
                        success = False
                        task.status = TaskStatus.FAILED
                        task.error = str(exc)
                        save_task(task)
                        self._emit(f"Lá»—i Scheduler á»Ÿ task {task.task_id}: {exc}", "ERROR", mission_id, task.task_id)
                    pending.discard(task.task_id)
                    failed_any = failed_any or not success
        return not failed_any and all(task.status == TaskStatus.COMPLETED for task in tasks)

    def _execute_task(self, task: Task, mission_id: str) -> bool:
        task.status = TaskStatus.RUNNING

            save_task(task)
            self._emit(task.error, "ERROR", mission_id, task.task_id)
            return False
        params = dict(task.input_data)
        params.setdefault("description", task.description)
        params["task_id"] = task.task_id
        try:
            result = skill.execute(params)
            import json
            log_execution(task.task_id, "SkillRunner", skill.manifest.name,
                          json.dumps(params, ensure_ascii=False, default=str)[:4000],
                          result.status.value, result.output)
            task.result = result.output
            task.artifact_path = result.evidence or task.artifact_path
            if result.status == ExecutionStatus.SUCCESS:
                task.status = TaskStatus.COMPLETED

    @classmethod

        if cls._skills and not force:
            return cls._skills
        cls._skills.clear()
        cls._errors.clear()
        root = str(ROOT_DIR)
        if root not in sys.path:
            sys.path.insert(0, root)
        SKILLS_DIR.mkdir(parents=True, exist_ok=True)
        for folder in sorted(SKILLS_DIR.iterdir()):
            manifest_path = folder / "manifest.json"
            module_path = folder / "skill.py"
            if not folder.is_dir() or not manifest_path.is_file() or not module_path.is_file():
                continue
            try:
                manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
                if manifest.get("enabled", True) is False:
                    continue
                module = importlib.import_module(f"skills.{folder.name}.skill")
                importlib.reload(module)
                candidates = [
                    getattr(module, name) for name in dir(module)
                    if isinstance(getattr(module, name), type)
                    and issubclass(getattr(module, name), BaseSkill)
                    and getattr(module, name) is not BaseSkill
                    and getattr(module, name).__module__ == module.__name__
                ]
                if len(candidates) != 1:
                    raise ValueError(f"Cáº§n Ä‘Ãºng 1 class BaseSkill trong {folder.name}, tÃ¬m tháº¥y {len(candidates)}")
                instance = candidates[0]()
                declared_name = manifest.get("name")
                if declared_name and declared_name != instance.manifest.name:
                    raise ValueError(f"TÃªn manifest '{declared_name}' khÃ´ng khá»›p skill '{instance.manifest.name}'")

                    raise ValueError(f"TrÃ¹ng tÃªn skill: {instance.manifest.name}")
                cls._skills[instance.manifest.name] = instance
            except Exception as exc:
                cls._errors.append(f"{folder.name}: {exc}")
        return cls._skills

                "description": skill.manifest.description,
                "input_schema": skill.manifest.input_schema,

class SecurityViolation(Exception):

class PermissionEngine:
    @staticmethod
    def validate_path(relative_path: str | Path) -> Path:
        raw = Path(relative_path)
        if raw.is_absolute():
            raise SecurityViolation("Chá»‰ cháº¥p nháº­n Ä‘Æ°á»ng dáº«n tÆ°Æ¡ng Ä‘á»‘i bÃªn trong workspace.")
        root = WORKSPACE_DIR.resolve()
        target = (root / raw).resolve()
        try:
            target.relative_to(root)
        except ValueError as exc:
            raise SecurityViolation(f"Tá»« chá»‘i truy cáº­p ngoÃ i workspace: {target}") from exc
        return target

    @staticmethod
    def ensure_parent(relative_path: str | Path) -> Path:
        target = PermissionEngine.validate_path(relative_path)
        target.parent.mkdir(parents=True, exist_ok=True)
        # Resolve again after mkdir to guard against unexpected symlink traversal.
        return PermissionEngine.validate_path(target.relative_to(WORKSPACE_DIR.resolve()))

from __future__ import annotations
import json
import sqlite3
from contextlib import contextmanager
from datetime import datetime, timezone
from typing import Any
from config import DB_PATH, STORAGE_DIR


def _connect() -> sqlite3.Connection:
    conn = sqlite3.connect(str(DB_PATH), timeout=30)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA busy_timeout = 30000")
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


@contextmanager
def _db():
    conn = _connect()
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def init_db() -> None:
    STORAGE_DIR.mkdir(parents=True, exist_ok=True)
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    with _db() as conn:
        conn.execute("PRAGMA journal_mode = WAL")
        conn.execute("""CREATE TABLE IF NOT EXISTS missions (
            mission_id TEXT PRIMARY KEY, request TEXT NOT NULL, status TEXT NOT NULL,
            summary TEXT DEFAULT '', created_at TEXT NOT NULL, updated_at TEXT NOT NULL
        )""")
        conn.execute("""CREATE TABLE IF NOT EXISTS tasks (
            task_id TEXT PRIMARY KEY, title TEXT, description TEXT, status TEXT,
            task_type TEXT, skill_name TEXT, dependencies TEXT, artifact_path TEXT,
            created_at TEXT
        )""")
        columns = {r[1] for r in conn.execute("PRAGMA table_info(tasks)").fetchall()}
        migrations = {
            "mission_id": "TEXT DEFAULT ''", "input_data": "TEXT DEFAULT '{}'",
            "result": "TEXT DEFAULT ''", "error": "TEXT DEFAULT ''",
        }
        for name, declaration in migrations.items():
            if name not in columns:
                conn.execute(f"ALTER TABLE tasks ADD COLUMN {name} {declaration}")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_tasks_mission ON tasks(mission_id)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_tasks_created ON tasks(created_at)")
        conn.execute("""CREATE TABLE IF NOT EXISTS events (
            id INTEGER PRIMARY KEY AUTOINCREMENT, timestamp TEXT NOT NULL,
            mission_id TEXT DEFAULT '', task_id TEXT DEFAULT '', level TEXT DEFAULT 'INFO',
            message TEXT NOT NULL
        )""")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_events_timestamp ON events(timestamp)")
        conn.execute("""CREATE TABLE IF NOT EXISTS executions (
            id INTEGER PRIMARY KEY AUTOINCREMENT, task_id TEXT, timestamp TEXT,
            agent TEXT, tool_name TEXT, arguments TEXT, status TEXT, output TEXT
        )""")
        conn.execute("""CREATE TABLE IF NOT EXISTS checkpoints (
            id INTEGER PRIMARY KEY AUTOINCREMENT, task_id TEXT, step_name TEXT,
            completed_at TEXT, artifact_path TEXT
        )""")
        conn.execute("""CREATE TABLE IF NOT EXISTS processes (
            pid INTEGER PRIMARY KEY, task_id TEXT, command TEXT, status TEXT, started_at TEXT
        )""")
        conn.execute("""CREATE TABLE IF NOT EXISTS approvals (
            id INTEGER PRIMARY KEY AUTOINCREMENT, task_id TEXT, action TEXT,
            reason TEXT, timestamp TEXT
        )""")


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def save_mission(mission_id: str, request: str, status: str, summary: str = "") -> None:
    now = _now()
    with _db() as conn:
        conn.execute("""INSERT INTO missions(mission_id,request,status,summary,created_at,updated_at)
            VALUES(?,?,?,?,?,?) ON CONFLICT(mission_id) DO UPDATE SET
            status=excluded.status, summary=excluded.summary, updated_at=excluded.updated_at""",
            (mission_id, request, status, summary, now, now))


def save_task(task: Any) -> None:
    with _db() as conn:
        conn.execute("""INSERT INTO tasks
            (task_id,title,description,status,task_type,skill_name,dependencies,artifact_path,created_at,mission_id,input_data,result,error)
            VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?)
            ON CONFLICT(task_id) DO UPDATE SET title=excluded.title, description=excluded.description,
            status=excluded.status, task_type=excluded.task_type, skill_name=excluded.skill_name,
            dependencies=excluded.dependencies, artifact_path=excluded.artifact_path,

def _load_dotenv(path: Path) -> None:
    if not path.exists():
        return

    for raw in path.read_text(encoding="utf-8").splitlines():
        line = raw.strip()

        if not line or line.startswith("#") or "=" not in line:
            continue

        key, value = line.split("=", 1)
        key, value = key.strip(), value.strip()

        if value.startswith(("'", '"')) and value.endswith(("'", '"')):
            value = value[1:-1]

        os.environ.setdefault(key, value)
